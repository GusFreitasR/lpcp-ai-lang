<program> → <stmt_list>

<stmt_list> → <stmt>
            | <stmt> <stmt_list>

<stmt> → <func_def>
       | <model_def>
       | <enum_def>
       | <const_decl> ";"
       | <assign> ";"
       | <func_call> ";"
       | <if_stmt>
       | <while_stmt>
       | <for_stmt>
       | <return_stmt> ";"
       | <print_stmt> ";"

<func_def> → "def" <id> "(" <param_list> ")" <block>
           | "def" <id> "(" ")" <block>

<param_list> → <id>
             | <id> "," <param_list>

<model_def> → "model" <id> <block>

<enum_def> → "enum" <id> "{" <id_list> "}"

<id_list> → <id>
          | <id> "," <id_list>

<const_decl> → "const" <id> "=" <const_val>

<const_val> → <literal>
            | <id>

<block> → "{" <stmt_list> "}"
        | "{" "}"

<assign> → <var> <assign_op> <expr>

<assign_op> → "="
            | "+="
            | "-="
            | "*="
            | "/="

<var> → <id>
      | <var> "[" <index_list> "]"

<func_call> → <id> "(" <arg_list> ")"
            | <id> "(" ")"
            | <built_in>

<built_in> → "input" "(" ")"
           | "len" "(" <expr> ")"
           | "split" "(" <expr> "," <expr> ")"
           | "zeros" "(" <arg_list> ")"
           | "ones" "(" <arg_list> ")"
           | "reshape" "(" <expr> "," <expr> ")"
           | "dot" "(" <expr> "," <expr> ")"
           | "to_int" "(" <expr> ")"
           | "to_float" "(" <expr> ")"
           | "to_string" "(" <expr> ")"
           | "to_bool" "(" <expr> ")"

<arg_list> → <arg>
           | <arg> "," <arg_list>

<arg> → <expr>
      | <id> "=" <expr>

<return_stmt> → "return" <expr>
              | "return"

<print_stmt> → "print" "(" <expr> ")"

<if_stmt> → "if" "(" <logic_expr> ")" <block>
          | "if" "(" <logic_expr> ")" <block> "else" <block>
          | "if" "(" <logic_expr> ")" <block> "else" <if_stmt>

<while_stmt> → "while" "(" <logic_expr> ")" <block>

<for_stmt> → "for" "(" <assign> ";" <logic_expr> ";" <assign> ")" <block>
           | "for" "(" <id> "in" <expr> ")" <block>

<logic_expr> → <logic_expr> "or" <logic_term>
             | <logic_term>

<logic_term> → <logic_term> "and" <rel_expr>
             | <rel_expr>

<rel_expr> → "not" <rel_expr>
           | <expr> <rel_op> <expr>
           | <expr>

<rel_op> → "=="
         | "!="
         | "<"
         | "<="
         | ">"
         | ">="
         | "in"

<expr> → <expr> "+" <term>
       | <expr> "-" <term>
       | <expr> "cat" <term>
       | <term>

<term> → <term> "*" <factor>
       | <term> "/" <factor>
       | <term> "%" <factor>
       | <term> "@" <factor>
       | <factor>

<factor> → <unary> "**" <factor>
         | <unary> "^" <factor>
         | <unary>

<unary> → "-" <unary>
        | "+" <unary>
        | <primary>

<primary> → <var>
          | <literal>
          | <func_call>
          | <list>
          | <list_comp>
          | <dict>
          | "(" <expr> ")"

<index_list> → <slice>
             | <slice> "," <index_list>

<slice> → <expr>
        | <expr> ":" <expr>
        | <expr> ":"
        | ":" <expr>
        | ":"

<list> → "[" <arg_list> "]"
       | "[" "]"

<list_comp> → "[" <expr> "for" <id> "in" <expr> "]"
            | "[" <expr> "for" <id> "in" <expr> "if" <logic_expr> "]"

<dict> → "{" <kv_list> "}"
       | "{" "}"

<kv_list> → <kv_pair>
          | <kv_pair> "," <kv_list>

<kv_pair> → <string_literal> ":" <expr>

<literal> → <int_literal>
          | <float_literal>
          | <string_literal>
          | <bool_literal>

<bool_literal> → "true"
               | "false"
